/* bind : bind creates a copy of a function and permanently locks it to a specific this value, so you can call it later. example const tv = { brand: "Sony", turnOn: function() { console.log(`${this.brand} TV is ON`); } } const remote = tv.turnOn; remote(); // ❌ 'this' is lost → "undefined TV is ON" // BIND creates a permanent remote for Sony TV const sonyRemote = tv.turnOn.bind(tv); sonyRemote(); // ✅ Works anytime! → "Sony TV is ON" */
